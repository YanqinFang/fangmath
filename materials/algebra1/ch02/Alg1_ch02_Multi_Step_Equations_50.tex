\section*{Level 5 \;—\; Multi-step with grouping (\#41--50)}
\begin{multicols}{2}
\begin{enumerate}\setcounter{enumi}{40}
  \item $3(2x - 4) - 5(x - 1) = 7$
  \item $2(x + 3) - 4(x - 1) = -6$
  \item $-(x - 3) + 2(2x + 1) = 4x + 8$
  \item $5x - 3(2 - x) = 4(x + 1) - 6$
  \item $4(x - 1) - 2(3x + 2) = -10$
  \item $2\!\left[\,3 - (x + 1)\,\right] = x - 4$
  \item $3\!\left[\,2x - (x + 1)\,\right] = 5(x - 3) - 2$
  \item $\dfrac{1}{2}(x - 4) + \dfrac{1}{3}(x + 6) = 7$
  \item $\dfrac{3}{4}(2x - 1) - \dfrac{1}{2}(x + 5) = \dfrac{1}{4}(x + 3)$
  \item $2(x + 3) - \bigl[\,3 - 2(x - 1)\,\bigr] = 3(x + 1)$
\end{enumerate}
\end{multicols}

\bigskip
\noindent\textbf{Tips for multi-step equations.}
\begin{enumerate}[label=\arabic*., leftmargin=2em, itemsep=2pt, topsep=2pt]
  \item Clear parentheses by distributing.
  \item Clear fractions by multiplying both sides by the LCD; clear decimals by multiplying by a power of $10$.
  \item Combine like terms on each side.
  \item Move variable terms to one side, constants to the other.
  \item Divide (or multiply) to isolate $x$.
  \item Substitute your answer back into the original equation to check.
\end{enumerate}
