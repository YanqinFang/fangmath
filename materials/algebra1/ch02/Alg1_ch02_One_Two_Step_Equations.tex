\sectionbar{Worked Examples}

\begin{multicols}{2}

\noindent\textbf{One-Step Example:}\quad Solve $\;x + 6 = 10$

\vspace{4pt}
\noindent Subtract 6 from both sides:
\[
  x + 6 - 6 = 10 - 6
\]
\[
  \boxed{x = 4}
\]
\noindent\textit{Check:} $4 + 6 = 10$ \checkmark

\columnbreak

\noindent\textbf{Two-Step Example:}\quad Solve $\;3x - 5 = 7$

\vspace{4pt}
\noindent Step \textcircled{1} Add 5 to both sides:
\[
  3x - 5 + 5 = 7 + 5 \;\Rightarrow\; 3x = 12
\]
\noindent Step \textcircled{2} Divide both sides by 3:
\[
  \frac{3x}{3} = \frac{12}{3} \;\Rightarrow\; \boxed{x = 4}
\]
\noindent\textit{Check:} $3(4)-5 = 12-5 = 7$ \checkmark

\end{multicols}
