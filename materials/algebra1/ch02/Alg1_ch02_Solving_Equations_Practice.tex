\banner{brandred}{PART 1 --- ONE-STEP EQUATIONS (10 problems)}

\begin{examplebox}
Solve for $x$: \quad $x + 7 = 12$
\begin{align*}
x + 7 &= 12 &&\text{Given equation}\\
x + 7 - 7 &= 12 - 7 &&\text{Subtract 7 from each side (undo the addition)}\\
x &= 5 &&\text{Simplify}
\end{align*}
\textbf{Remember:} addition and subtraction are inverse operations; multiplication and division are inverse operations. Undo whatever is being done to $x$.
\end{examplebox}

\banner{brandred}{PART 2 --- TWO-STEP EQUATIONS (10 problems)}

\begin{examplebox}
Solve for $x$: \quad $3x + 4 = 19$
\begin{align*}
3x + 4 &= 19 &&\text{Given equation}\\
3x + 4 - 4 &= 19 - 4 &&\text{Subtract 4 from each side (undo the addition)}\\
3x &= 15 &&\text{Simplify}\\
\dfrac{3x}{3} &= \dfrac{15}{3} &&\text{Divide each side by 3 (undo the coefficient)}\\
x &= 5 &&\text{Simplify}
\end{align*}
\textbf{Remember:} undo addition/subtraction \emph{first}, then undo multiplication/division.
\end{examplebox}

\banner{brandred}{PART 3 --- MULTI-STEP EQUATIONS WITH FRACTIONS \& DECIMALS (10 problems)}

\begin{examplebox}
Solve for $x$: \quad $2(x - 3) + 1.5 = 9.5$
\begin{align*}
2(x-3) + 1.5 &= 9.5 &&\text{Given equation}\\
2x - 6 + 1.5 &= 9.5 &&\text{Distribute the 2}\\
2x - 4.5 &= 9.5 &&\text{Combine like terms}\\
2x - 4.5 + 4.5 &= 9.5 + 4.5 &&\text{Add 4.5 to each side}\\
2x &= 14 &&\text{Simplify}\\
x &= 7 &&\text{Divide each side by 2}
\end{align*}
\end{examplebox}

\begin{tipbox}
\textbf{Fractions:} multiply both sides by the denominator (or the LCD, if there is more than one fraction) to clear it before doing anything else. \quad\textbf{Decimals:} treat them just like whole numbers --- add, subtract, multiply, and divide the same way, just watch your decimal point.
\end{tipbox}

\banner{brandred}{PART 4 --- WORD PROBLEMS (5 problems)}

\begin{examplebox}
The sum of a number and 8 is 23. Find the number.
\begin{align*}
\text{Let } x &= \text{the unknown number.}\\
x + 8 &= 23 &&\text{Translate the sentence into an equation}\\
x + 8 - 8 &= 23 - 8 &&\text{Subtract 8 from each side}\\
x &= 15 &&\text{Simplify}
\end{align*}
\textbf{The number is 15.}\\[1mm]
\textbf{Tip:} underline the key phrase, define what $x$ represents in words, write the equation, solve it, then answer the original question in a full sentence.
\end{examplebox}
