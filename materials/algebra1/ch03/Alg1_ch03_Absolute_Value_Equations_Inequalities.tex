%%  KEY CONCEPTS
%% ════════════════════════════════════════════════════════════
\begin{keybox}
\begin{tabular}{@{}ll@{}}
\textbf{Absolute Value Equation} &
  $|x|=a \;\Rightarrow\; x=a \text{ or } x=-a$\\[6pt]
\textbf{Absolute Value Inequality} $(\leq)$ &
  $|x|\leq a \;\Rightarrow\; -a\leq x\leq a$ \quad (AND, closed interval)\\[4pt]
\textbf{Absolute Value Inequality} $(\geq)$ &
  $|x|\geq a \;\Rightarrow\; x\leq -a \text{ or } x\geq a$ \quad (OR, union)\\[6pt]
\multicolumn{2}{@{}l@{}}{\textbf{Verbal meaning:} $|x-c|\leq d$ means
  ``the distance between $x$ and $c$ is at most $d$.''}
\end{tabular}

\vspace{4pt}
\textbf{Steps to Solve:}
\begin{enumerate}[leftmargin=1.8em,topsep=2pt,itemsep=1pt]
  \item Isolate the absolute value expression.
  \item Split into two cases (positive and negative).
  \item Solve each case and check your answers.
  \item Graph the solution on a number line.
\end{enumerate}
\end{keybox}

\vspace{8pt}

%% ════════════════════════════════════════════════════════════
%%  EXAMPLE 1 — EQUATION
%% ════════════════════════════════════════════════════════════
\begin{exbox}{Example 1 \textemdash\ Equation:\quad $|f-6|=3$}
\textbf{Step 1: Split into two cases}
\[f-6=3 \qquad\text{or}\qquad f-6=-3\]
\textbf{Step 2: Solve each}
\[f=9 \qquad\text{or}\qquad f=3\]
\textbf{Step 3: Check}\quad
$f=9$: $|9-6|=3$ \checkmark\qquad
$f=3$: $|3-6|=3$ \checkmark\\[2pt]
\textbf{Solution:} $f=3$ or $f=9$

\numline{0}{10}{0.7cm}{%
  \node[dot,label=above:{$3$}] at (3,0){};
  \node[dot,label=above:{$9$}] at (9,0){};
}
\end{exbox}

\vspace{6pt}

%% ════════════════════════════════════════════════════════════
%%  EXAMPLE 2 — INEQUALITY ≤
%% ════════════════════════════════════════════════════════════
\begin{exbox}{Example 2 \textemdash\ Inequality:\quad $|f-6|\leq 3$}
\textbf{Method (Algebraic):} Apply $-3\leq f-6\leq 3$, then add 6:
\[3\leq f\leq 9\]
\textbf{Check:}\quad $f=4$: $|4-6|=2\leq 3$ \checkmark\qquad
$f=10$: $|10-6|=4\leq 3$ \texttimes\ (non-solution, correct)

\numline{0}{10}{0.7cm}{%
  \node[dot,label=above:{$3$}] at (3,0){};
  \node[dot,label=above:{$9$}] at (9,0){};
  \draw[blue!70!black,line width=1.8pt] (3,0)--(9,0);
}
\end{exbox}

\vspace{6pt}

%% ════════════════════════════════════════════════════════════
%%  EXAMPLE 3 — INEQUALITY >
%% ════════════════════════════════════════════════════════════
\begin{exbox}{Example 3 \textemdash\ Inequality:\quad $5|b+8|-7>13$}
\textbf{Step 1: Isolate}\quad
$5|b+8|>20 \;\Rightarrow\; |b+8|>4$\\[4pt]
\textbf{Step 2: Split (OR)}
\[b+8<-4 \qquad\text{or}\qquad b+8>4\]
\[b<-12 \qquad\text{or}\qquad b>-4\]

\numline{-14}{-2}{0.65cm}{%
  \node[odot,label=above:{$-12$}] at (-12,0){};
  \node[odot,label=above:{$-4$}] at (-4,0){};
  \draw[blue!70!black,line width=1.8pt,{Stealth[length=5pt]}-]
    (-12,0)--(-14.6,0);
  \draw[blue!70!black,line width=1.8pt,-{Stealth[length=5pt]}]
    (-4,0)--(-1.4,0);
}
\end{exbox}

%% ── Section A ────────────────────────────────────────────
\begin{tcolorbox}[anskey, title={Section A \textemdash\ Absolute Value Equations}]
\begin{enumerate}[leftmargin=1.8em, itemsep=4pt]
  \item $|x|=5$ \hfill $\boldsymbol{x=-5}$ \textbf{ or } $\boldsymbol{x=5}$
  \item $|x-3|=7$ \hfill $\boldsymbol{x=-4}$ \textbf{ or } $\boldsymbol{x=10}$
  \item $|2x+1|=9$ \hfill $\boldsymbol{x=-5}$ \textbf{ or } $\boldsymbol{x=4}$
  \item $|3x-6|=12$ \hfill $\boldsymbol{x=-2}$ \textbf{ or } $\boldsymbol{x=6}$
  \item $|x+4|-2=5 \;\Rightarrow\; |x+4|=7$ \hfill $\boldsymbol{x=-11}$ \textbf{ or } $\boldsymbol{x=3}$
\end{enumerate}
\end{tcolorbox}

\vspace{8pt}

%% ── Section B ────────────────────────────────────────────
\begin{tcolorbox}[anskey, title={Section B \textemdash\ Inequalities $(\leq\;/\;<)$}]
\begin{enumerate}[leftmargin=1.8em, itemsep=4pt, start=6]
  \item $|x|\leq 4$ \hfill $\boldsymbol{-4\leq x\leq 4}$
  \item $|x-2|<5 \;\Rightarrow\; -5<x-2<5$ \hfill $\boldsymbol{-3<x<7}$
  \item $|2x-4|\leq 6 \;\Rightarrow\; -6\leq 2x-4\leq 6$ \hfill $\boldsymbol{-1\leq x\leq 5}$
  \item $|3x+3|<9 \;\Rightarrow\; -9<3x+3<9$ \hfill $\boldsymbol{-4<x<2}$
  \item $2|x+1|-3\leq 7 \;\Rightarrow\; |x+1|\leq 5$ \hfill $\boldsymbol{-6\leq x\leq 4}$
  \item $3|x-5|+1\leq 10 \;\Rightarrow\; |x-5|\leq 3$ \hfill $\boldsymbol{2\leq x\leq 8}$
\end{enumerate}
\end{tcolorbox}

\vspace{8pt}

%% ── Section C ────────────────────────────────────────────
\begin{tcolorbox}[anskey, title={Section C \textemdash\ Inequalities $(\geq\;/\;>)$}]
\begin{enumerate}[leftmargin=1.8em, itemsep=4pt, start=12]
  \item $|x|>3$ \hfill $\boldsymbol{x<-3}$ \textbf{ or } $\boldsymbol{x>3}$
  \item $|x+2|\geq 4 \;\Rightarrow\; x+2\leq -4$ or $x+2\geq 4$ \hfill $\boldsymbol{x\leq -6}$ \textbf{ or } $\boldsymbol{x\geq 2}$
  \item $|2x-6|>4 \;\Rightarrow\; 2x-6<-4$ or $2x-6>4$ \hfill $\boldsymbol{x<1}$ \textbf{ or } $\boldsymbol{x>5}$
  \item $3|x-1|\geq 9 \;\Rightarrow\; |x-1|\geq 3$ \hfill $\boldsymbol{x\leq -2}$ \textbf{ or } $\boldsymbol{x\geq 4}$
  \item $5|b+8|-7>13 \;\Rightarrow\; |b+8|>4$ \hfill $\boldsymbol{b<-12}$ \textbf{ or } $\boldsymbol{b>-4}$
\end{enumerate}
\end{tcolorbox}

\vspace{8pt}

%% ── Section D ────────────────────────────────────────────
\begin{tcolorbox}[anskey, title={Section D \textemdash\ Mixed \& Word Problems}]
\begin{enumerate}[leftmargin=1.8em, itemsep=6pt, start=17]
  \item $|4x-8|\geq 12$\\
        $4x-8\leq -12 \;\Rightarrow\; x\leq -1$\quad or\quad
        $4x-8\geq 12 \;\Rightarrow\; x\geq 5$\hfill
        $\boldsymbol{x\leq -1}$ \textbf{ or } $\boldsymbol{x\geq 5}$

  \item Graph shows closed dots at $-3$ and $3$, shaded between.\\
        \textbf{Inequality:} $\boldsymbol{|x|\leq 3}$\quad
        Solution: $\boldsymbol{-3\leq x\leq 3}$\\
        Check: $x=0$: $|0|=0\leq 3$ \checkmark\quad
               $x=4$: $|4|=4\leq 3$ \texttimes\

  \item $|T-70|\leq 3 \;\Rightarrow\; -3\leq T-70\leq 3$\hfill
        $\boldsymbol{67^\circ\text{F}\leq T\leq 73^\circ\text{F}}$

  \item $|2x-1|=|x+4|$\\
        \textit{Case 1:} $2x-1=x+4 \;\Rightarrow\; x=5$\\
        \textit{Case 2:} $2x-1=-(x+4) \;\Rightarrow\; 3x=-3 \;\Rightarrow\; x=-1$\hfill
        $\boldsymbol{x=-1}$ \textbf{ or } $\boldsymbol{x=5}$
\end{enumerate}
