\noindent\fbox{%
  \begin{minipage}{\dimexpr\textwidth-2\fboxsep-2\fboxrule\relax}
  \vspace{4pt}
  \textbf{Goal:} Get $x$ alone on one side by undoing what is done to it.\\[6pt]
  \begin{tabular}{@{}p{2.8cm} p{5cm} p{5.5cm}@{}}
  \textbf{Form} & \textbf{Undo with\ldots} & \textbf{Quick Example}\\[2pt]
  $x + a = b$ & subtract $a$ from both sides & $x+4=9 \Rightarrow x=5$\\[2pt]
  $x - a = b$ & add $a$ to both sides        & $x-3=8 \Rightarrow x=11$\\[2pt]
  $ax = b$    & divide both sides by $a$      & $4x=20 \Rightarrow x=5$\\[2pt]
  $\frac{x}{a}=b$ & multiply both sides by $a$ & $\frac{x}{3}=5 \Rightarrow x=15$
  \end{tabular}
  \vspace{4pt}
  \end{minipage}
}

\topictitle{Topic 2 \quad Solving Absolute Value Equations}

\noindent\fbox{%
  \begin{minipage}{\dimexpr\textwidth-2\fboxsep-2\fboxrule\relax}
  \vspace{4pt}
  \textbf{Key Fact:} $|x|$ = distance from $x$ to $0$. Distance is never negative.\\[5pt]
  \begin{tabular}{@{}p{2.2cm} l@{}}
  $c > 0$: & $|E| = c \;\Longrightarrow\; E = c \;\text{ or }\; E = -c$ \quad (two solutions)\\[3pt]
  $c = 0$: & $|E| = 0 \;\Longrightarrow\; E = 0$ \quad (one solution)\\[3pt]
  $c < 0$: & $|E| = c \;\Longrightarrow\;$ \textbf{No solution}
  \end{tabular}\\[5pt]
  \textbf{3 Steps:}\quad
  \textcircled{1}~Isolate $|\cdot|$\quad
  \textcircled{2}~Split: $E = c$ \;or\; $E = -c$\quad
  \textcircled{3}~Solve each equation
  \vspace{4pt}
  \end{minipage}
}

\vspace{10pt}

%%── Type 1: |x| = c ────────────────────────────────────────────────────────
\examplebox{%
\textbf{Type 1 Example} (Basic): Solve $|x| = 4$\\[4pt]
Already isolated. Split: $x = 4$ \;or\; $x = -4$\\[2pt]
\textit{Think of it on a number line: 4 units left of 0 and 4 units right of 0.}
\[
  \boxed{x = 4 \quad\text{or}\quad x = -4}
\]
}

%%── Type 2: |x ± a| = c ────────────────────────────────────────────────────
\examplebox{%
\textbf{Type 2 Example} (Shift): Solve $|x - 3| = 5$\\[4pt]
Already isolated. Split:\\[2pt]
$x - 3 = 5 \;\Rightarrow\; x = 8$
\qquad or \qquad
$x - 3 = -5 \;\Rightarrow\; x = -2$
\[
  \boxed{x = 8 \quad\text{or}\quad x = -2}
\]
\textit{Check: $|8-3|=|5|=5$ \checkmark \quad $|-2-3|=|-5|=5$ \checkmark}
}

%%── Type 3: Isolate first ───────────────────────────────────────────────────
\examplebox{%
\textbf{Type 3 Example} (Isolate First): Solve $|x + 4| + 2 = 9$\\[4pt]
\textbf{Step \textcircled{1}} Subtract 2 from both sides: $|x+4| = 7$\\[3pt]
\textbf{Step \textcircled{2}} Split: $x+4=7 \;\Rightarrow\; x=3$
\quad or \quad $x+4=-7 \;\Rightarrow\; x=-11$
\[
  \boxed{x = 3 \quad\text{or}\quad x = -11}
\]
}

\textbf{Case 2:} $2x-1=-7 \;\Rightarrow\; 2x=-6 \;\Rightarrow\; x=-3$
\[
  \boxed{x = 4 \quad\text{or}\quad x = -3}

\topictitle{Topic 3 \quad Solving Inequalities}

\noindent\fbox{%
  \begin{minipage}{\dimexpr\textwidth-2\fboxsep-2\fboxrule\relax}
  \vspace{4pt}
  \textbf{Inequality Symbols:}
  \quad $<$ less than \quad
  $>$ greater than \quad
  $\leq$ less than or equal to \quad
  $\geq$ greater than or equal to\\[5pt]
  \textbf{Rules:} Same as equations EXCEPT ---\\
  \hspace*{1cm}$\bullet$ Multiply or divide by a \textbf{positive} number: symbol stays the same.\\
  \hspace*{1cm}$\bullet$ Multiply or divide by a \textbf{negative} number: \textbf{FLIP the symbol!}\\[4pt]
  \textit{Example}: $-2x > 6 \;\div(-2)\;\Rightarrow\; x < -3$ \quad (symbol flipped from $>$ to $<$)
  \vspace{4pt}
  \end{minipage}
}
