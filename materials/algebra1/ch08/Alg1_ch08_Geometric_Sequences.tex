\begin{keyidea}
A \textbf{geometric sequence} multiplies by the same number every time. That number is the \textbf{common ratio} $r$.
\[ r = \frac{\text{any term}}{\text{the term before it}} = \frac{a_2}{a_1} \]
\begin{center}\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|l|l|l|}\hline
\rowcolor{fmkey!12}\textbf{Formula} & \textbf{Looks like} & \textbf{What it tells you}\\\hline
\textbf{Recursive} & $a_1 = \text{first term}, \quad a_n = r\cdot a_{n-1}$ & how to get the \emph{next} term from the one before\\\hline
\textbf{Explicit} & $a_n = a_1\cdot r^{\,n-1}$ & how to jump straight to \emph{any} term $a_n$\\\hline
\end{tabular}\end{center}
\end{keyidea}
\begin{mistakebox}
Divide to find $r$, do not subtract: \ for $3, 6, 12, \ldots$ \ $r = 6 \div 3 = 2$ (not $6 - 3 = 3$).\par
The exponent is $n - 1$, not $n$: \ $a_1 = a_1\cdot r^{0}$. \ And a recursive formula \textbf{must} state $a_1$.
\end{mistakebox}
\sectionbar{Part 1 \quad Write the Recursive Formula}{Problems 1--20}
\begin{watchme}
\textbf{Write a recursive formula for} \ $5,\ 15,\ 45,\ 135,\ \ldots$\par\vspace{1mm}
\renewcommand{\arraystretch}{1.6}\begin{tabular}{@{}l l@{}}
\textbf{Step 1} \ Find $r$: divide a term by the one before it & $r = 15 \div 5 = 3$ \ (check: $45 \div 15 = 3$ \checkmark)\\
\textbf{Step 2} \ Write the first term & $a_1 = 5$\\
\textbf{Step 3} \ Each term is $r$ times the one before & $a_n = 3\cdot a_{n-1}$\\
\textbf{Answer} & $\boldsymbol{a_1 = 5, \quad a_n = 3\cdot a_{n-1}}$
\end{tabular}\par\vspace{1mm}
\textit{Negative ratio?} \ For $2, -6, 18, \ldots$: \ $r = -6 \div 2 = -3$, \ so $a_1 = 2$, \ $a_n = (-3)\cdot a_{n-1}$ \ (the signs alternate).
\end{watchme}

\sectionbar{Part 2 \quad Write the Explicit Formula}{Problems 21--40}
\begin{watchme}
\textbf{Write an explicit formula for} \ $4,\ 8,\ 16,\ 32,\ \ldots$\par\vspace{1mm}
\renewcommand{\arraystretch}{1.6}\begin{tabular}{@{}l l@{}}
\textbf{Step 1} \ Find $a_1$ and $r$ & $a_1 = 4$, \quad $r = 8 \div 4 = 2$\\
\textbf{Step 2} \ Substitute into $a_n = a_1\cdot r^{\,n-1}$ & $\boldsymbol{a_n = 4\cdot 2^{\,n-1}}$\\
\textbf{Check} \ $n = 3$ & $a_3 = 4\cdot 2^{2} = 16$ \ \checkmark
\end{tabular}\par\vspace{1mm}
\textit{Fraction or negative ratio?} Put it in parentheses: \ $81, 27, 9, \ldots$ gives $a_n = 81\cdot\left(\tfrac13\right)^{n-1}$; \ $1, -2, 4, \ldots$ gives $a_n = (-2)^{n-1}$.
\end{watchme}
