\section*{\textcolor{brandpurple}{1. Summary}}
\sectionrule

\noindent\textbf{Pattern A --- Perfect Square Trinomial.}
\[
a^2 + 2ab + b^2 = (a+b)^2 \qquad\text{and}\qquad a^2 - 2ab + b^2 = (a-b)^2 .
\]
The first and last terms are perfect squares ($a^2,\ b^2$) and the middle term is $2ab$.

\medskip
\noindent\textbf{Pattern B --- Difference of Squares.}
\[
a^2 - b^2 = (a-b)(a+b).
\]
Two perfect-square terms separated by a minus sign, with no middle term.

\medskip
\noindent\textbf{Pattern C --- Factoring by Grouping} (four terms).
\begin{enumerate}[label=(\arabic*),leftmargin=2em,itemsep=1pt]
  \item Group the four terms into two pairs.
  \item Factor out the GCF of the \emph{first} two terms and the GCF of the \emph{last} two terms.
  \item If both pairs now share the \emph{same} binomial factor, factor it out.
\end{enumerate}
\[
ax + ay + bx + by = a(x+y) + b(x+y) = (x+y)(a+b).
\]
\textit{Always check whether a resulting factor can be factored further (e.g.\ a difference of squares).}

\begin{center}
\fbox{$(a+b)^2 = a^2 + 2ab + b^2$}\quad
\fbox{$a^2 - b^2 = (a-b)(a+b)$}\quad
\fbox{$ax+ay+bx+by=(x+y)(a+b)$}
\end{center}

\section*{\textcolor{brandpurple}{2. Worked Examples}}
\sectionrule

\noindent\textbf{\textcolor{brandpurple}{Part A --- Perfect Square Trinomials}}

\medskip
\textbf{A1.}\quad $x^2 - 12x + 36$: \; product $36$, sum $-12 \Rightarrow -6,-6$.
\[ x^2 - 12x + 36 = x^2 - 6x - 6x + 36 = (x-6)^2 . \]
$\begin{array}{c|c|c}
   & x & -6 \\ \hline
 x & x^2 & -6x \\ \hline
 -6 & -6x & 36
\end{array}$

\medskip
\textbf{A2.}\quad $x^2 + 6x + 9 \Rightarrow 3,3$:\quad $x^2+6x+9=(x+3)^2$.
\qquad
\textbf{A3.}\quad $x^2 - 14x + 49 \Rightarrow -7,-7$:\quad $x^2-14x+49=(x-7)^2$.

\bigskip
\noindent\textbf{\textcolor{brandteal}{Part B --- Difference of Squares}}

\medskip
\textbf{B1.}\quad $x^2 - 9 = x^2 + 0\cdot x - 9$; \; product $-9$, sum $0 \Rightarrow 3,-3$.
\[ x^2 - 9 = x^2 + 3x - 3x - 9 = (x-3)(x+3) . \]
$\begin{array}{c|c|c}
   & x & -3 \\ \hline
 x & x^2 & -3x \\ \hline
 3 & 3x & -9
\end{array}$
\qquad
\textbf{B2.}\quad $x^2 - 100 = (x+10)(x-10)$.

\bigskip
\noindent\textbf{\textcolor{brandblue}{Part C --- Factoring by Grouping}}

\medskip
\textbf{C.}\quad Factor $2z^3 + 6z^2 + 3z + 9$.
\begin{align*}
2z^3 + 6z^2 + 3z + 9 &= (2z^3 + 6z^2) + (3z + 9) &&\text{group into pairs}\\
 &= 2z^2(z+3) + 3(z+3) &&\text{GCF of each pair}\\
 &= (z+3)(2z^2 + 3) &&\text{factor out the common } (z+3)
\end{align*}

\section*{\textcolor{brandteal}{4. Detailed Answer Key}}
\sectionrule

\noindent\textbf{\textcolor{brandpurple}{Part A --- Perfect Square Trinomials}}
\begin{enumerate}[label=\textbf{A\arabic*.},leftmargin=2.6em,itemsep=5pt]
  \item $a=x,\ b=4$; $2ab=8x$ \checkmark. $\boxed{(x+4)^2}$
  \item $a=x,\ b=5$; middle $-10x$ \checkmark. $\boxed{(x-5)^2}$
  \item $a=x,\ b=10$; $2ab=20x$ \checkmark. $\boxed{(x+10)^2}$
  \item $a=x,\ b=1$; middle $-2x$ \checkmark. $\boxed{(x-1)^2}$
  \item $a=2x,\ b=3$; $2ab=12x$ \checkmark. $\boxed{(2x+3)^2}$
  \item $a=3x,\ b=5$; middle $-30x$ \checkmark. $\boxed{(3x-5)^2}$
  \item $a=x,\ b=\tfrac12$; $2ab=x$ \checkmark. $\boxed{\left(x+\tfrac{1}{2}\right)^2}$
  \item $a=4x,\ b=5$; middle $-40x$ \checkmark. $\boxed{(4x-5)^2}$
  \item $a=5x,\ b=6$; $2ab=60x$ \checkmark. $\boxed{(5x+6)^2}$
  \item $16m^2-72m+81=(4m-9)^2$, so side $=\boxed{\,4m-9\ \text{ft}\,}$.
\end{enumerate}

\medskip
\noindent\textbf{\textcolor{brandteal}{Part B --- Difference of Squares}}
\begin{enumerate}[label=\textbf{B\arabic*.},leftmargin=2.6em,itemsep=5pt]
  \item $\boxed{(x-5)(x+5)}$ \item $\boxed{(x-7)(x+7)}$ \item $\boxed{(x-1)(x+1)}$
  \item $\boxed{(2x-3)(2x+3)}$ \item $\boxed{(3x-4)(3x+4)}$ \item $\boxed{(5x-8)(5x+8)}$
  \item $\boxed{(x-12)(x+12)}$ \item $\boxed{(7x-10)(7x+10)}$
  \item $\boxed{\left(x-\tfrac{2}{3}\right)\left(x+\tfrac{2}{3}\right)}$
  \item Remaining area $=x^2-64=\boxed{(x-8)(x+8)}$.
\end{enumerate}

\medskip
\noindent\textbf{\textcolor{brandblue}{Part C --- Factoring by Grouping}}\\
\textit{Shown: GCF of first pair, GCF of second pair, then the common binomial.}
\begin{enumerate}[label=\textbf{C\arabic*.},leftmargin=2.6em,itemsep=5pt]
  \item $3a^2(a+4)+2(a+4)=\boxed{(a+4)(3a^2+2)}$
  \item $2b^2(3b-5)+3(3b-5)=\boxed{(3b-5)(2b^2+3)}$
  \item $4c^2(c+5)-3(c+5)=\boxed{(c+5)(4c^2-3)}$
  \item $2d^2(5d+2)+3(5d+2)=\boxed{(5d+2)(2d^2+3)}$
  \item $7n^2(n+3)+2(n+3)=\boxed{(n+3)(7n^2+2)}$
  \item $5t^2(t-3)+4(t-3)=\boxed{(t-3)(5t^2+4)}$
  \item $4x^2(3x+2)+5(3x+2)=\boxed{(3x+2)(4x^2+5)}$
  \item $3u^2(3u-2)+5(3u-2)=\boxed{(3u-2)(3u^2+5)}$
  \item $4s^2(2s+3)-5(2s+3)=\boxed{(2s+3)(4s^2-5)}$
  \item $3j^2(2j-3)-4(2j-3)=\boxed{(2j-3)(3j^2-4)}$
  \item $2e^2(e+3)-5(e+3)=\boxed{(e+3)(2e^2-5)}$
  \item $4f^2(f+2)-9(f+2)=(f+2)(4f^2-9)=\boxed{(f+2)(2f-3)(2f+3)}$ \; \small(diff.\ of squares!)
  \item $25g^2(g-2)-1(g-2)=(g-2)(25g^2-1)=\boxed{(g-2)(5g-1)(5g+1)}$ \; \small(diff.\ of squares!)
\end{enumerate}
