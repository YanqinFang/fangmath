\section*{Completing the Square \& Equations (Multiple Choice)}

\begin{enumerate}

\item Rewrite in vertex form:
\[
y = x^2 + 6x + 2
\]
\begin{enumerate}
\item $y = (x+3)^2 + 7$
\item $y = (x-3)^2 - 7$
\item $y = (x+3)^2 - 7$
\item $y = (x+6)^2 - 34$
\end{enumerate}

\item Solve by completing the square:
\[
x^2 - 4x + 1 = 0
\]
\begin{enumerate}
\item $x = 2 \pm \sqrt{5}$
\item $x = -2 \pm \sqrt{5}$
\item $x = 4 \pm \sqrt{3}$
\item $x = 2 \pm 3$
\end{enumerate}

\item Rewrite in vertex form:
\[
y = 2x^2 + 8x + 3
\]
\begin{enumerate}
\item $y = 2(x+4)^2 - 29$
\item $y = 2(x-2)^2 - 5$
\item $y = 2(x+2)^2 + 3$
\item $y = 2(x+2)^2 - 5$
\end{enumerate}

\item Solve:
\[
x^2 + 10x + 1 = 0
\]
\begin{enumerate}
\item $x = -5 \pm \sqrt{24}$
\item $x = 5 \pm \sqrt{24}$
\item $x = -10 \pm 1$
\item $x = 5 \pm 24$
\end{enumerate}

\item Rewrite in vertex form:
\[
y = 3x^2 - 12x + 7
\]
\begin{enumerate}
\item $y = 3(x+2)^2 - 5$
\item $y = 3(x-2)^2 + 7$
\item $y = 3(x-2)^2 - 5$
\item $y = 3(x+2)^2 + 7$
\end{enumerate}

\item Solve by completing the square:
\[
x^2 + 8x + 20 = 0
\]
\begin{enumerate}
\item $x = -4 \pm 2$
\item $x = -4 \pm 2i$
\item $x = 4 \pm 2i$
\item $x = -8 \pm 4$
\end{enumerate}

\item Rewrite:
\[
y = -x^2 + 6x - 2
\]
\begin{enumerate}
\item $y = -(x+3)^2 + 7$
\item $y = -(x-3)^2 + 7$
\item $y = -(x-3)^2 - 11$
\item $y = -(x+6)^2 + 34$
\end{enumerate}

\item Solve:
\[
2x^2 + 4x - 3 = 0
\]
\begin{enumerate}
\item $x = -1 \pm \sqrt{2}$
\item $x = 1 \pm \sqrt{2}$
\item $x = -1 \pm \frac{\sqrt{10}}{2}$
\item $x = 1 \pm \frac{\sqrt{10}}{2}$
\end{enumerate}

\end{enumerate}
