{\color{fmtitle}\LARGE\bfseries Midpoint \& Distance Formula --- Practice\ifnum\iskey=1 \ --- Answer Key\fi}\par\vspace{1mm}
{\color{fmgray}A worked example comes before each type. Midpoint = Mean, Distance = Difference. Show your thinking.}\par\vspace{3mm}
\nameline

\needspace{6cm}\topicbar{Type 1 \quad Midpoint on a Number Line}
\begin{watchme}Find the midpoint of $-6$ and $10$.\par
Midpoint = mean: $\dfrac{-6 + 10}{2} = \dfrac{4}{2}$. \quad \textbf{Answer: $\boldsymbol{2}$}\end{watchme}
\pair{\sdp{1}{Find the midpoint of $4$ and $18$.}{$11$}}{\sdp{2}{Find the midpoint of $-9$ and $3$.}{$-3$}}
\pair{\sdp{3}{Find the midpoint of $-7$ and $-1$.}{$-4$}}{\sdp{4}{Find the midpoint of $2.5$ and $8.5$.}{$5.5$}}

\needspace{6cm}\topicbar{Type 2 \quad Midpoint in the Coordinate Plane}
\begin{watchme}$A(-2, 5)$ and $B(6, 1)$. Find the midpoint $M$.\par
$x$: $\dfrac{-2 + 6}{2} = 2$, \quad $y$: $\dfrac{5 + 1}{2} = 3$. \quad \textbf{Answer: $\boldsymbol{M(2, 3)}$}\end{watchme}
\pair{\sdp{5}{$(1, 4)$ and $(7, 10)$}{$(4, 7)$}}{\sdp{6}{$(-3, -2)$ and $(5, 6)$}{$(1, 2)$}}
\pair{\sdp{7}{$(0, -5)$ and $(-8, 3)$}{$(-4, -1)$}}{\sdp{8}{$(2, 7)$ and $(5, -2)$}{$(3.5,\ 2.5)$}}

\needspace{6.5cm}\topicbar{Type 3 \quad Find the Other Endpoint}
\begin{watchme}$M(3, 4)$ is the midpoint of $\overline{AB}$, and $A(1, 1)$. Find $B$.\par
From $A$ to $M$: $x$ goes up $2$, $y$ goes up $3$. Do it again from $M$: $(3 + 2,\ 4 + 3)$.\par
(Or: $B = (2 \cdot 3 - 1,\ 2 \cdot 4 - 1)$.) \quad \textbf{Answer: $\boldsymbol{B(5, 7)}$}\end{watchme}
\pair{\sdp{9}{Midpoint $M(4, 6)$, endpoint $A(2, 1)$. Find $B$.}{$B(6, 11)$}}{\sdp{10}{Midpoint $M(0, 0)$, endpoint $A(-3, 5)$. Find $B$.}{$B(3, -5)$}}
\pair{\sdp{11}{Midpoint $M(-1, 2)$, endpoint $A(3, 8)$. Find $B$.}{$B(-5, -4)$}}{\sdp{12}{On a number line, $M$ is at $7$ and $A$ is at $2$. Where is $B$?}{$12$}}

\needspace{6.5cm}\topicbar{Type 4 \quad Distance Formula}
\begin{watchme}$A(1, 1)$ and $B(7, 9)$. Find $AB$.\par
Differences: $7 - 1 = 6$, \ $9 - 1 = 8$. \quad $AB = \sqrt{6^2 + 8^2} = \sqrt{100}$. \quad \textbf{Answer: $\boldsymbol{10}$}\end{watchme}
\pair{\sdp{13}{$(2, 3)$ and $(5, 7)$}{$5$}}{\sdp{14}{$(-1, -2)$ and $(11, 3)$}{$13$}}
\pair{\sdp{15}{$(0, 4)$ and $(6, -4)$}{$10$}}{\sdp{16}{\LH\ $(1, 2)$ and $(4, 5)$. Give the exact answer and a decimal.}{$3\sqrt{2} \approx 4.24$}}
