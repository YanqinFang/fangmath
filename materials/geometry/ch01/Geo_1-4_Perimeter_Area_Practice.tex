\needspace{6cm}\topicbar{Type 1 \quad Horizontal \& Vertical Lengths}
\begin{watchme}Find the length from $(-2, 3)$ to $(5, 3)$.\par
Same $y$, so subtract the $x$'s: $5 - (-2) = 7$. \quad \textbf{Answer: $\boldsymbol{7}$}\end{watchme}
\pair{\sdp{1}{$(1, 4)$ to $(9, 4)$}{$8$}}{\sdp{2}{$(3, -2)$ to $(3, 6)$}{$8$}}
\pair{\sdp{3}{$(-5, -1)$ to $(2, -1)$}{$7$}}{\sdp{4}{$(-4, 7)$ to $(-4, -3)$}{$10$}}

\needspace{6cm}\topicbar{Type 2 \quad Rectangles: Perimeter \& Area}
\begin{watchme}Rectangle $A(-1, 1)$, $B(5, 1)$, $C(5, 5)$, $D(-1, 5)$.\par
Width $= 5 - (-1) = 6$, height $= 5 - 1 = 4$. \ $P = 2(6 + 4) = 20$, \ $A = 6 \cdot 4 = 24$.\par
\textbf{Answer: $\boldsymbol{P = 20}$ units, $\boldsymbol{A = 24}$ square units}\end{watchme}
\pair{\sdpw{5}{$(0, 0)$, $(7, 0)$, $(7, 3)$, $(0, 3)$}{$P = 20,\ A = 21$}}{\sdpw{6}{$(-2, -1)$, $(4, -1)$, $(4, 4)$, $(-2, 4)$}{$P = 22,\ A = 30$}}
\pair{\sdpw{7}{Square $(1, 1)$, $(6, 1)$, $(6, 6)$, $(1, 6)$}{$P = 20,\ A = 25$}}{\sdpw{8}{Rectangle $(0, 0)$, $(9, 0)$, $(9, k)$, $(0, k)$ has area $54$. Find $k$ $(k > 0)$.}{$k = 6$}}

\needspace{6.5cm}\topicbar{Type 3 \quad Area of a Triangle}
\begin{watchme}Triangle $(0, 0)$, $(8, 0)$, $(3, 5)$.\par
The base lies on the $x$-axis: $b = 8$. The height is straight up to $(3, 5)$: $h = 5$.\par
$A = \tfrac12 \cdot 8 \cdot 5$. \quad \textbf{Answer: $\boldsymbol{20}$ square units}\end{watchme}
\pair{\sdp{9}{$(1, 1)$, $(7, 1)$, $(4, 6)$}{$15$}}{\sdp{10}{$(-2, 0)$, $(6, 0)$, $(0, 7)$}{$28$}}
\pair{\sdp{11}{$(0, 0)$, $(0, 6)$, $(5, 2)$ \ {\small(hint: the base is vertical)}}{$15$}}{\sdp{12}{\LH\ $(2, -1)$, $(10, -1)$, $(12, 4)$}{$20$}}

\needspace{6.5cm}\topicbar{Type 4 \quad Slanted Sides: Use Pythagoras}
\begin{watchme}Triangle $(0, 0)$, $(3, 0)$, $(0, 4)$. Find the perimeter and area.\par
Legs $3$ and $4$; slanted side $\sqrt{3^2 + 4^2} = 5$. \ $P = 3 + 4 + 5 = 12$, \ $A = \tfrac12 \cdot 3 \cdot 4 = 6$.\par
\textbf{Answer: $\boldsymbol{P = 12,\ A = 6}$}\end{watchme}
\pair{\sdpw{13}{$(0, 0)$, $(5, 0)$, $(0, 12)$: find the perimeter.}{$P = 30$}}{\sdpw{14}{$(1, 1)$, $(7, 1)$, $(1, 9)$: find the perimeter and area.}{$P = 24,\ A = 24$}}
\pair{\sdpw{15}{$(-3, 0)$, $(3, 0)$, $(0, 4)$: find the perimeter and area.}{$P = 16,\ A = 12$}}{\sdpw{16}{\LH\ Square $(0, 0)$, $(3, 4)$, $(-1, 7)$, $(-4, 3)$: find the perimeter and area.}{$P = 20,\ A = 25$}}
