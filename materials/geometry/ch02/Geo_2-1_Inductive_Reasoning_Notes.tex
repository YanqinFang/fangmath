\vid{What is inductive reasoning?}
\begin{keyidea}\textbf{Inductive reasoning}: look at examples, find a \fb[2cm]{pattern}, and use it to \fb[2cm]{predict}.\par\vspace{2mm}
\textbf{1.} Look \quad $\to$ \quad \textbf{2.} Describe the \fb[1.6cm]{rule} \quad $\to$ \quad \textbf{3.} Predict\end{keyidea}

\vid{A number pattern}
\begin{watchme}{\large $3,\ 7,\ 11,\ 15,\ \ldots$}\par\vspace{2mm}
\renewcommand{\arraystretch}{1.7}\begin{tabular}{@{}l l@{}}
Rule: \fb[2cm]{add 4} each time & check \textbf{every} pair: $7 - 3$, $11 - 7$, $15 - 11$\\
Next term: $15 + 4 = $ \fb[1.2cm]{$19$} & \\
10th term: $3 + 9 \cdot 4 = $ \fb[1.2cm]{$39$} & start at $3$, add $4$ \fb[1.4cm]{nine} times\\
\end{tabular}\end{watchme}

\vid{A picture pattern}
\begin{keyidea}\begin{minipage}[c]{0.5\linewidth}\centering
\begin{tikzpicture}\foreach \n/\x in {1/0,2/1.0,3/2.5,4/4.4}{\foreach \i in {1,...,\n}{\foreach \j in {1,...,\n}{\fill[fmkey] (\x+\i*0.3,\j*0.3) circle(0.09);}}}\end{tikzpicture}\par
$1,\ 4,\ 9,\ 16,\ \ldots$\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}These are \fb[3.4cm]{square numbers}: $1^2, 2^2, 3^2, 4^2$.\par\vspace{2mm}
Next: $5^2 = $ \fb[1.2cm]{$25$}\end{minipage}\end{keyidea}

\vid{Conjecture and counterexample}
\begin{keyidea}A \fb[2.4cm]{conjecture} is a statement you believe is true, based on a pattern.\par\vspace{2mm}
Conjecture: ``The sum of two numbers is always greater than each number.''\par\vspace{2mm}
A \fb[3cm]{counterexample}: $-2 + 5 = $ \fb[1cm]{$3$}, and $3 < 5$. \ So the conjecture is \fb[1.4cm]{false}.\par\vspace{1mm}
{\small One counterexample is enough to show a conjecture is false.}\end{keyidea}

\vid{Watch out}
\begin{mistakebox}$2,\ 4,\ 8,\ \ldots$ \quad Times 2? \ Next is \fb[1cm]{$16$}. \quad Or $+2, +4, +6$? \ Next is \fb[1cm]{$14$}.\par\vspace{1mm}
Two rules can fit the same first terms --- that's why inductive reasoning gives a \fb[2.4cm]{conjecture}, not a proof.\end{mistakebox}

\vid{Your turn}
\begin{yourturn}Find the next two terms: \quad {\large $1,\ 3,\ 6,\ 10,\ \ldots$}\par\vspace{2mm}
The differences are $+2, +3, +4$, so next \fb[1cm]{$+5$}, then \fb[1cm]{$+6$}.\par\vspace{1mm}
$10 + 5 = $ \fb[1.2cm]{$15$}, \qquad $15 + 6 = $ \fb[1.2cm]{$21$}\end{yourturn}

\begin{notebox}\textbf{Remember:}\quad Look $\to$ describe the rule $\to$ predict.\quad Check the rule on \textbf{every} pair.\quad A pattern gives a \textbf{conjecture}; one \textbf{counterexample} shows it is false.\end{notebox}
