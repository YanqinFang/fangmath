\vid{If $p$, then $q$}
\begin{keyidea}{\large If \colorbox{fmkey!15}{it rains}, then \colorbox{fmturn!25}{the ground is wet}.}\par\vspace{2mm}
$p$ = the \fb[2.4cm]{hypothesis} (the ``if'' part) \qquad $q$ = the \fb[2.4cm]{conclusion} (the ``then'' part)\par\vspace{1mm}
Written: \ \fb[1.6cm]{$p \to q$}\end{keyidea}

\vid{Four related statements}
\begin{keyidea}\renewcommand{\arraystretch}{1.8}\begin{tabular}{@{}l l l@{}}
\textbf{Conditional} & $p \to q$ & If it rains, then the ground is wet.\\
\textbf{Converse} & \fb[1.6cm]{$q \to p$} & \fb[1.6cm]{switch}: If the ground is wet, then it rained.\\
\textbf{Inverse} & \fb[1.6cm]{$\sim p \to \sim q$} & \fb[1.6cm]{negate}: If it does not rain, then the ground is not wet.\\
\textbf{Contrapositive} & \fb[1.6cm]{$\sim q \to \sim p$} & \fb[3cm]{switch and negate}: If the ground is not wet, then it did not rain.\\
\end{tabular}\end{keyidea}

\vid{True or false?}
\begin{watchme}\renewcommand{\arraystretch}{1.6}\begin{tabular}{@{}l l@{}}
Conditional: If it rains, the ground is wet. & \fb[1.4cm]{true}\\
Converse: If the ground is wet, it rained. & \fb[1.4cm]{false} \ --- counterexample: a \fb[2cm]{sprinkler}\\
Contrapositive: If the ground is not wet, it did not rain. & \fb[1.4cm]{true}\\
\end{tabular}\end{watchme}

\vid{The rule}
\begin{keyidea}A conditional and its \fb[3cm]{contrapositive} are always both true or both false.\par\vspace{1mm}
The \fb[2cm]{converse} and the \fb[2cm]{inverse} also match each other --- but they may not match the original.\end{keyidea}

\vid{Biconditional}
\begin{keyidea}If a conditional \textbf{and} its converse are both true, write ``\fb[3cm]{if and only if}''.\par\vspace{1mm}
Example: An angle is right if and only if it measures \fb[1.2cm]{$90^\circ$}.\end{keyidea}

\vid{Your turn}
\begin{yourturn}``If an angle measures $30^\circ$, then it is acute.'' \ Write the \textbf{converse}. Is it true?\par\vspace{2mm}
Converse: If an angle is \fb[1.6cm]{acute}, then it measures \fb[1.2cm]{$30^\circ$}.\par\vspace{1mm}
True or false? \fb[1.4cm]{false} \qquad Counterexample: a \fb[1.4cm]{$50^\circ$} angle is acute. {\small (any acute angle except $30^\circ$)}\end{yourturn}

\begin{notebox}\textbf{Remember:}\quad Converse \textbf{switches}, inverse \textbf{negates}, contrapositive does \textbf{both}.\quad A conditional and its contrapositive are both true or both false.\quad Both directions true? Use ``if and only if.''\end{notebox}
