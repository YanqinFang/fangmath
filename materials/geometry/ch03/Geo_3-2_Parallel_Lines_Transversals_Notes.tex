\vid{When the lines are parallel}
\begin{keyidea}\begin{minipage}[c]{0.36\linewidth}\centering\tfig\end{minipage}\hfill
\begin{minipage}[c]{0.6\linewidth}If $\ell \parallel m$:\par\vspace{1mm}
Corresponding angles are \fb[2.4cm]{congruent}.\par\vspace{1mm}
Alternate interior and alternate exterior angles are \fb[2.4cm]{congruent}.\par\vspace{1mm}
Same-side interior angles are \fb[3cm]{supplementary} (add to \fb[1.2cm]{$180^\circ$}).\end{minipage}\end{keyidea}

\vid{The easy way to see it}
\begin{keyidea}Every angle is either \fb[1.6cm]{acute} or \fb[1.6cm]{obtuse}.\par\vspace{1mm}
Two acute angles are \fb[1.4cm]{equal}. \ Two obtuse angles are \fb[1.4cm]{equal}. \ Acute $+$ obtuse $=$ \fb[1.4cm]{$180^\circ$}.\end{keyidea}

\vid{Find the angle}
\begin{watchme}$\ell \parallel m$, \ $m\angle 1 = 110^\circ$. \ Find $m\angle 5$ and $m\angle 6$.\par\vspace{1mm}
\renewcommand{\arraystretch}{1.6}\begin{tabular}{@{}l l@{}}
$\angle 1$ and $\angle 5$ are \fb[3cm]{corresponding} & $m\angle 5 = $ \fb[1.4cm]{$110^\circ$}\\
$\angle 5$ and $\angle 6$ are a \fb[2.4cm]{linear pair} & $m\angle 6 = 180 - 110 = $ \fb[1.4cm]{$70^\circ$}\\
\end{tabular}\end{watchme}

\vid{With algebra}
\begin{watchme}Alternate interior angles: $(3x + 15)^\circ$ and $(5x - 25)^\circ$.\par\vspace{1mm}
\renewcommand{\arraystretch}{1.6}\begin{tabular}{@{}l l@{}}
$3x + 15 = 5x - 25$ & alternate interior angles are \fb[2.4cm]{congruent}\\
$40 = 2x$, \ $x = $ \fb[1.2cm]{$20$} & \\
angle $= 3(20) + 15 = $ \fb[1.4cm]{$75^\circ$} & answer the question they asked\\
\end{tabular}\end{watchme}

\vid{Watch out}
\begin{mistakebox}Same-side interior angles are \fb[3cm]{supplementary}, not equal. Set them to add to \fb[1.4cm]{$180^\circ$}.\end{mistakebox}

\vid{Your turn}
\begin{yourturn}$\ell \parallel m$. Same-side interior angles are $(2x + 10)^\circ$ and $(3x - 5)^\circ$. Find $x$ and both angles.\par\vspace{2mm}
$(2x + 10) + (3x - 5) = $ \fb[1.2cm]{$180$} \ $\Rightarrow$ \ $5x + 5 = 180$ \ $\Rightarrow$ \ $x = $ \fb[1.2cm]{$35$}\par\vspace{1mm}
Angles: \fb[1.4cm]{$80^\circ$} and \fb[1.4cm]{$100^\circ$}\end{yourturn}

\begin{notebox}\textbf{Remember:}\quad Parallel lines: corresponding and alternate angles are \textbf{congruent}; same-side interior angles are \textbf{supplementary}.\quad Acute $=$ acute, obtuse $=$ obtuse, acute $+$ obtuse $= 180^\circ$.\end{notebox}
