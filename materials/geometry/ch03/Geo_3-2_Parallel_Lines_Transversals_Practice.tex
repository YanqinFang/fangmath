\needspace{7cm}\topicbar{Type 1 \quad Find the Angles}
\begin{watchme}\begin{minipage}[c]{0.3\linewidth}\centering\tfig\end{minipage}\hfill
\begin{minipage}[c]{0.66\linewidth}$\ell \parallel m$ and $m\angle 2 = 65^\circ$. \ Find $m\angle 7$.\par
$\angle 2$ and $\angle 7$ are alternate exterior, so they are congruent. \quad \textbf{Answer: $\boldsymbol{65^\circ}$.}\par
Find $m\angle 8$: $\angle 7$ and $\angle 8$ are a linear pair, so $180 - 65 = 115^\circ$.\end{minipage}\end{watchme}
In every problem $\ell \parallel m$ (figure above).\par

\needspace{6cm}\topicbar{Type 2 \quad Solve With Algebra: Congruent Pairs}
\begin{watchme}Corresponding angles: $(4x - 10)^\circ$ and $(2x + 30)^\circ$. \ Congruent: $4x - 10 = 2x + 30 \Rightarrow 2x = 40 \Rightarrow x = 20$.\par
Angle: $4(20) - 10 = 70^\circ$. \quad \textbf{Answer: $\boldsymbol{x = 20,\ 70^\circ}$}\end{watchme}
\needspace{4.6cm}\noindent\algp{7}{Alternate interior angles: $(5x - 8)^\circ$ and $(3x + 22)^\circ$. Find $x$ and the angle.}{$x = 15$, $67^\circ$}\hfill\algp{8}{Alternate exterior angles: $(7x + 4)^\circ$ and $(9x - 20)^\circ$. Find $x$ and the angle.}{$x = 12$, $88^\circ$}\par

\needspace{6cm}\topicbar{Type 3 \quad Solve With Algebra: Supplementary Pairs}
\begin{watchme}Same-side interior angles: $(3x + 20)^\circ$ and $(2x + 10)^\circ$. \ They add to $180^\circ$: $5x + 30 = 180 \Rightarrow x = 30$.\par
Angles: $3(30) + 20 = 110^\circ$ and $2(30) + 10 = 70^\circ$. \ Check: $110 + 70 = 180$ \ding{51}\end{watchme}
\needspace{4.6cm}\noindent\algp{9}{Same-side interior angles: $(4x + 12)^\circ$ and $(2x - 6)^\circ$. Find both angles.}{$x = 29$: $128^\circ$ and $52^\circ$}\hfill\algp{10}{$m\angle 1 = (6x + 4)^\circ$ and $m\angle 7 = (2x + 16)^\circ$. Find $x$.}{$x = 20$}\par

\needspace{6cm}\topicbar{Type 4 \quad Two Steps}
\begin{watchme}$m\angle 1 = (x + 40)^\circ$ and $m\angle 6 = (2x + 20)^\circ$. \ $\angle 1 \cong \angle 5$ (corresponding), and $\angle 5$, $\angle 6$ are a linear pair, so $(x + 40) + (2x + 20) = 180 \Rightarrow x = 40$.\end{watchme}
\needspace{4.6cm}\noindent\algp{11}{$m\angle 2 = (3x)^\circ$ and $m\angle 5 = (x + 20)^\circ$. Find $x$.}{$x = 40$}\hfill\algp{12}{\LH\ $m\angle 4 = (x^2)^\circ$ and $m\angle 5 = (5x + 14)^\circ$, $x > 0$. Find $m\angle 4$.}{$x = 7$, $49^\circ$}\par
