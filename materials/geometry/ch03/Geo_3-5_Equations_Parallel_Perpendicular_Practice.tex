\needspace{6cm}\topicbar{Type 1 \quad Parallel and Perpendicular Slopes}
\begin{watchme}$y = -\tfrac45 x + 2$ has slope $-\tfrac45$. \ Parallel slope: $-\tfrac45$. \ Perpendicular slope: flip and change the sign, $\tfrac54$.\end{watchme}
Give the parallel slope and the perpendicular slope.\par
\needspace{3.4cm}\noindent\sdp{1}{$y = 3x - 7$}{$3$; \ $-\tfrac13$}\hfill\sdp{2}{$y = -\tfrac25 x + 1$}{$-\tfrac25$; \ $\tfrac52$}\par\vspace{2mm}
\needspace{3.4cm}\noindent\sdp{3}{$y = -x + 4$}{$-1$; \ $1$}\hfill\sdp{4}{$4x + 2y = 10$}{$-2$; \ $\tfrac12$}\par

\needspace{6cm}\topicbar{Type 2 \quad Parallel, Perpendicular or Neither?}
\begin{watchme}$y = 2x + 1$ and $y = -\tfrac12 x - 3$: \ slopes $2$ and $-\tfrac12$ multiply to $-1$. \quad \textbf{Answer: perpendicular.}\end{watchme}
\needspace{3.4cm}\noindent\sdp{5}{$y = 5x - 2$ and $y = 5x + 9$}{parallel}\hfill\sdp{6}{$y = \tfrac34 x$ and $y = -\tfrac43 x + 1$}{perpendicular}\par\vspace{2mm}
\needspace{3.4cm}\noindent\sdp{7}{$y = 2x + 3$ and $y = \tfrac12 x - 1$}{neither}\hfill\sdp{8}{$x + 3y = 6$ and $y = 3x - 2$}{perpendicular}\par

\needspace{6cm}\topicbar{Type 3 \quad Write a Parallel Line}
\begin{watchme}Through $(1, -4)$, parallel to $y = -2x + 5$. \ Same slope $-2$: $y + 4 = -2(x - 1) \Rightarrow y = -2x - 2$.\end{watchme}
\needspace{4.6cm}\noindent\algp{9}{Through $(3, 1)$, parallel to $y = 2x - 6$.}{$y = 2x - 5$}\hfill\algp{10}{Through $(-2, 5)$, parallel to $y = -\tfrac12 x + 3$.}{$y = -\tfrac12 x + 4$}\par

\needspace{6cm}\topicbar{Type 4 \quad Write a Perpendicular Line}
\begin{watchme}Through $(2, 7)$, perpendicular to $y = \tfrac23 x - 1$. \ Slope $-\tfrac32$: $y - 7 = -\tfrac32(x - 2) \Rightarrow y = -\tfrac32 x + 10$.\end{watchme}
\needspace{4.6cm}\noindent\algp{11}{Through $(4, -1)$, perpendicular to $y = 2x + 3$.}{$y = -\tfrac12 x + 1$}\hfill\algp{12}{Through $(-3, 2)$, perpendicular to $y = -3x + 1$.}{$y = \tfrac13 x + 3$}\par\vspace{2mm}
\needspace{4.6cm}\noindent\algp{13}{Through $(0, 5)$, perpendicular to $x - 4y = 8$.}{$y = -4x + 5$}\hfill\algp{14}{\LH\ Find $k$ so that $y = kx + 1$ is perpendicular to $3x + 6y = 2$.}{$k = 2$}\par
