\parttitle{A}{Similar Rectangles}
\begin{keyidea}
Two shapes are \textbf{similar} when one is an enlargement (or shrink) of the other: \textbf{same shape, different size}.
For rectangles, \textbf{long side $\leftrightarrow$ long side} and \textbf{short side $\leftrightarrow$ short side}. The rectangles are similar when both ratios are the same number:
\[ \frac{\text{long side of 2nd}}{\text{long side of 1st}} \;=\; \frac{\text{short side of 2nd}}{\text{short side of 1st}} \;=\; \text{scale factor} \]
\end{keyidea}
\begin{watchme}\textbf{Example 1 --- Are they similar?}\par\vspace{2mm}\begin{minipage}[c]{0.42\linewidth}\centering \begin{tikzpicture}[baseline=0pt]\filldraw[fill=fmkey!12,draw=fmtitle,thick] (0,0) rectangle (0.840,1.400);\node[below] at (0.420,0) {$3$};\node[left] at (0,0.700) {$5$};\filldraw[fill=fmturn!18,draw=fmtitle,thick] (2.140,0) rectangle (3.820,2.800);\node[below] at (2.980,0) {$6$};\node[left] at (2.140,1.400) {$10$};\end{tikzpicture}\end{minipage}\hfill\begin{minipage}[c]{0.55\linewidth}Long sides: $\dfrac{10}{5} = 2$ \qquad Short sides: $\dfrac{6}{3} = 2$\par\vspace{1mm}Both ratios are $2$, so the rectangles \textbf{are similar}.\par Scale factor $=\mathbf{2}$.\end{minipage}\end{watchme}
\begin{watchme}\textbf{Example 2 --- Find the missing side.} \;The rectangles are similar.\par\vspace{2mm}\begin{minipage}[c]{0.42\linewidth}\centering \begin{tikzpicture}[baseline=0pt]\filldraw[fill=fmkey!12,draw=fmtitle,thick] (0,0) rectangle (0.373,0.933);\node[below] at (0.187,0) {$2$};\node[left] at (0,0.467) {$5$};\filldraw[fill=fmturn!18,draw=fmtitle,thick] (1.673,0) rectangle (2.793,2.800);\node[below] at (2.233,0) {$6$};\node[left] at (1.673,1.400) {$x$};\end{tikzpicture}\end{minipage}\hfill\begin{minipage}[c]{0.55\linewidth}{\color{fmwatch}\textbf{Method 1 --- Scale factor}}\par\vspace{1.5mm} Use the pair you know: \;$\dfrac{6}{2} = 3$, so the scale factor is $3$.\par\vspace{1mm} Multiply: \;$x = 5 \times 3 = \mathbf{15}$.\par\vspace{2mm}{\color{fmwatch}\textbf{Method 2 --- Proportion}}\par\vspace{1.5mm} Write $\dfrac{\text{2nd}}{\text{1st}}$ for both pairs of matching sides:\par\vspace{1mm}$\dfrac{6}{2} = \dfrac{x}{5}$ \;$\Rightarrow$\; $2x = 6 \times 5 = 30$ \;$\Rightarrow$\; $x = \mathbf{15}$\end{minipage}\end{watchme}
\begin{mistake}
$2 \times 3$ and $4 \times 5$ are \textbf{not} similar. Adding $2$ to every side changes the shape.
Similar shapes come from \textbf{multiplying} every side by the same number.
\end{mistake}

\parttitle{B}{Similar Triangles}
\begin{keyidea}
Match the sides by size: \textbf{shortest $\leftrightarrow$ shortest, middle $\leftrightarrow$ middle, longest $\leftrightarrow$ longest}.
The triangles are similar when \textbf{all three} ratios are the same number (the scale factor).\par\vspace{1mm}
In a \textbf{similarity statement} $\triangle ABC \sim \triangle DEF$, the \textbf{order} of the letters shows which vertices match: $A \leftrightarrow D$, $B \leftrightarrow E$, $C \leftrightarrow F$.
\end{keyidea}

\begin{mistake}
Checking only \textbf{two} pairs of sides is not enough. Sides $4, 6, 3$ and $8, 10, 6$ have two ratios equal to $2$,
but $\tfrac{10}{6} \neq 2$ --- so they are \textbf{not} similar. Check all three!
\end{mistake}
