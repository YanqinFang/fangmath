\parttitle{B}{Similar Triangles}
\leveltag{fmwatch}{EASY}\par\vspace{1mm}
\begin{enumerate}[label=\textbf{\arabic*.},start=1,itemsep=2mm,leftmargin=*]
\item $\dfrac{4}{2} = 2$, \; $\dfrac{6}{3} = 2$, \; $\dfrac{8}{4} = 2$ \;$\Rightarrow$\; \ans{$\triangle ABC \sim \triangle DEF$}, scale factor \ans{$2$}
\item $\dfrac{9}{3} = 3$, \; $\dfrac{15}{5} = 3$, \; $\dfrac{21}{7} = 3$ \;$\Rightarrow$\; \ans{$\triangle ABC \sim \triangle PQR$}, scale factor \ans{$3$}
\item $\dfrac{6}{4} = \tfrac{3}{2}$, \; $\dfrac{7}{5} = \tfrac{7}{5}$, \; $\dfrac{8}{6} = \tfrac{4}{3}$ \;$\Rightarrow$\; \ans{Not similar}
\item $\dfrac{10}{5} = 2$, \; $\dfrac{12}{6} = 2$, \; $\dfrac{16}{8} = 2$ \;$\Rightarrow$\; \ans{$\triangle ABC \sim \triangle JKL$}, scale factor \ans{$2$}
\item $\dfrac{6}{3} = 2$, \; $\dfrac{8}{4} = 2$, \; $\dfrac{10}{6} = \tfrac{5}{3}$ \;$\Rightarrow$\; \ans{Not similar}
\end{enumerate}
\begin{enumerate}[label=\textbf{\arabic*.},start=6,itemsep=2mm,leftmargin=*]
\item \textbf{M1:} $AB \leftrightarrow DE$: \;$\dfrac{12}{4} = 3$; \; $x = EF$ matches $BC = 5$: \;$x = 5 \times 3$ $=$ \ans{$15$}\par\textbf{M2:} $\dfrac{DE}{AB} = \dfrac{EF}{BC}$: \; $\dfrac{12}{4} = \dfrac{x}{5}$ \;$\Rightarrow$\; $4x = 12 \times 5 = 60$ \;$\Rightarrow$\; $x = 60 \div 4$ $=$ \ans{$15$}
\item \textbf{M1:} $AB \leftrightarrow PQ$: \;$\dfrac{8}{4} = 2$; \; $x = RP$ matches $CA = 3$: \;$x = 3 \times 2$ $=$ \ans{$6$}\par\textbf{M2:} $\dfrac{PQ}{AB} = \dfrac{RP}{CA}$: \; $\dfrac{8}{4} = \dfrac{x}{3}$ \;$\Rightarrow$\; $4x = 8 \times 3 = 24$ \;$\Rightarrow$\; $x = 24 \div 4$ $=$ \ans{$6$}
\item \textbf{M1:} $BC \leftrightarrow YZ$: \;$\dfrac{14}{7} = 2$; \; $x = XY$ matches $AB = 6$: \;$x = 6 \times 2$ $=$ \ans{$12$}\par\textbf{M2:} $\dfrac{YZ}{BC} = \dfrac{XY}{AB}$: \; $\dfrac{14}{7} = \dfrac{x}{6}$ \;$\Rightarrow$\; $7x = 14 \times 6 = 84$ \;$\Rightarrow$\; $x = 84 \div 7$ $=$ \ans{$12$}
\item \textbf{M1:} $AB \leftrightarrow JK$: \;$\dfrac{21}{7} = 3$; \; $x = KL$ matches $BC = 9$: \;$x = 9 \times 3$ $=$ \ans{$27$}\par\textbf{M2:} $\dfrac{JK}{AB} = \dfrac{KL}{BC}$: \; $\dfrac{21}{7} = \dfrac{x}{9}$ \;$\Rightarrow$\; $7x = 21 \times 9 = 189$ \;$\Rightarrow$\; $x = 189 \div 7$ $=$ \ans{$27$}
\item \textbf{M1:} $AB \leftrightarrow ST$: \;$\dfrac{16}{8} = 2$; \; $x = US$ matches $CA = 6$: \;$x = 6 \times 2$ $=$ \ans{$12$}\par\textbf{M2:} $\dfrac{ST}{AB} = \dfrac{US}{CA}$: \; $\dfrac{16}{8} = \dfrac{x}{6}$ \;$\Rightarrow$\; $8x = 16 \times 6 = 96$ \;$\Rightarrow$\; $x = 96 \div 8$ $=$ \ans{$12$}
\end{enumerate}
